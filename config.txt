# Gomoku AI Configuration File
# ====================================
# このファイルでAIの動作を設定できます

# 探索深度（1～8推奨）
# 深さ1: 瞬時（初級レベル）
# 深さ2-3: ~200ms（中級レベル）
# 深さ4-5: ~1-5秒（上級レベル）
# 深さ6-8: ~10-60秒（大会レベル）
SEARCH_DEPTH=5

# 置換表のサイズ（エントリ数、2のべき乗推奨）
# 大きいほど探索が速くなるが、メモリを消費
# 例: 65536 = 約1MB, 262144 = 約4MB, 1048576 = 約16MB
TRANSPOSITION_TABLE_SIZE=262144

# デバッグモード（0=OFF, 1=ON）
# ONにすると詳細なログが出力されます
DEBUG_MODE=0

# タイムアウト（ミリ秒、0=無制限）
# 設定した時間内に探索を打ち切ります
TIMEOUT_MS=10000

# 学習モード（PRACTICE=練習モード, PRODUCTION=本番モード）
# PRACTICE: 学習しながら学習データを参照（練習・トレーニング用）
# PRODUCTION: 学習せず学習データのみ参照（本番・大会用）
LEARNING_MODE=PRACTICE

# 反復深化探索（0=OFF, 1=ON）
# ONにすると深さ1から徐々に探索を深める
# メリット: 時間制限への対応、早期の良い手の発見、置換表の効率向上
ITERATIVE_DEEPENING=1

# 詳細デバッグ出力（0=OFF, 1=基本, 2=詳細, 3=超詳細）
# 0: デバッグ出力なし
# 1: 基本情報（探索開始・終了、最善手）
# 2: 詳細情報（各深さの結果、置換表統計）
# 3: 超詳細（全候補手の評価値、枝刈り情報）
DEBUG_LEVEL=1

# Iterative Deepening 時間制限閾値（0.0～1.0）
# この値 × TIMEOUT_MS の時間までに深度探索を開始します
# 例: 0.7 = 70%の時点で次の深度に進まない
#     0.8 = 80%の時点で次の深度に進まない（推奨）
#     0.9 = 90%の時点で次の深度に進まない（積極的）
# 注意: 探索中に時間切れになった場合、その深度の結果は破棄され、
#       前の深度の結果が返されます
ID_TIME_THRESHOLD=0.8

# 厳格な時間制限（ミリ秒、0=TIMEOUT_MSと同じ）
# TIMEOUT_MSとは別に、探索中に「絶対に超えてはいけない」時間制限を設定できます
# 0: TIMEOUT_MSと同じ時間制限を使用（デフォルト）
# >0: TIMEOUT_MSとは独立した厳格な制限時間
#
# 使用例:
#   TIMEOUT_MS=10000, STRICT_TIME_LIMIT_MS=0     → 10秒の通常制限
#   TIMEOUT_MS=10000, STRICT_TIME_LIMIT_MS=8000  → 通常10秒だが、絶対に8秒で停止
#   TIMEOUT_MS=15000, STRICT_TIME_LIMIT_MS=12000 → 通常15秒だが、絶対に12秒で停止
#
# メリット:
#   - ターン制ゲームで時間切れによる失格を防ぐ
#   - 制限時間ぎりぎりまで探索しつつ、安全マージンを確保
STRICT_TIME_LIMIT_MS=0

# 早期カットオフ時間閾値（0.0～1.0）
# Early cutoffを適用する最低時間使用率を設定します
# 0.0: 常にEarly cutoffを適用（高スコアなら即座に終了）
# 0.5: 時間を50%以上使った場合のみEarly cutoffを適用（推奨）
# 1.0: Early cutoffを実質無効化（時間を使い切るまで探索）
#
# 使用例:
#   EARLY_CUTOFF_TIME_THRESHOLD=0.0  → 高スコアなら即座に終了（時間を節約）
#   EARLY_CUTOFF_TIME_THRESHOLD=0.5  → 時間に余裕があれば継続探索（バランス型）
#   EARLY_CUTOFF_TIME_THRESHOLD=0.8  → ほぼ時間を使い切るまで探索（慎重型）
#
# 推奨設定:
#   - 序盤・中盤: 0.5～0.7（時間を有効活用）
#   - 終盤: 0.2～0.4（高スコアなら早期終了）
EARLY_CUTOFF_TIME_THRESHOLD=1.0
