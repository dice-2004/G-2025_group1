# ====================================================
# 五目並べAI 設定ファイル (config.txt)
# ====================================================
# このファイルはAIの動作を制御します。
# 各パラメータの意味と推奨値を示します。

# ----------------------------------------------------
# 探索設定
# ----------------------------------------------------

# SEARCH_DEPTH: AI の探索深度
# 値が大きいほど強くなりますが、思考時間も長くなります。
# 推奨値: 3～5
#   3: 高速（初心者向け）
#   4: 標準（中級者向け）
#   5: 強力（上級者向け）※思考時間が長い
SEARCH_DEPTH=4

# TIMEOUT_MS: 1手あたりの最大思考時間（ミリ秒）
# Iterative Deepening使用時に有効
# 推奨値: 5000～30000
#   5000: 高速対戦
#   10000: 標準対戦（10秒）
#   30000: じっくり対戦（30秒）
TIMEOUT_MS=10000

# ITERATIVE_DEEPENING: 反復深化探索を使用するか
# 0: 無効（固定深度探索）
# 1: 有効（時間制限内で最大深度まで探索）
# 推奨値: 1（有効）
ITERATIVE_DEEPENING=1

# ID_TIME_THRESHOLD: 次の深度に進む時間しきい値
# 現在の深度の探索時間が、残り時間のこの比率を超えたら探索を打ち切る
# 0.0～1.0の範囲（0.8 = 80%）
# 推奨値: 0.7～0.85
#   0.7: 安全（早めに打ち切り）
#   0.8: 標準
#   0.85: 攻撃的（ギリギリまで探索）
ID_TIME_THRESHOLD=0.80

# STRICT_TIME_LIMIT_MS: 厳格な時間制限（ミリ秒）
# TIMEOUT_MSとは別に、探索中に「絶対に超えてはいけない」時間制限を設定
# 0: TIMEOUT_MSと同じ時間制限を使用（デフォルト）
# >0: TIMEOUT_MSとは独立した厳格な制限時間
# 推奨値: 0 または TIMEOUT_MS の 80～90%
#   0: 通常の時間制限のみ使用
#   8000: TIMEOUT_MS=10000の場合、絶対に8秒で停止
#   27000: TIMEOUT_MS=30000の場合、絶対に27秒で停止
# 用途: ターン制ゲームで時間切れ失格を防ぐ安全マージン
STRICT_TIME_LIMIT_MS=0

# EARLY_CUTOFF_TIME_THRESHOLD: 早期カットオフ時間閾値（0.0～1.0）
# Early cutoffを適用する最低時間使用率を設定します。
# 高スコアが得られた場合でも、時間に余裕があれば探索を続けます。
#
# 値の意味:
#   0.0: 常にEarly cutoffを適用（高スコアなら即座に終了）
#   0.5: 時間を50%以上使った場合のみ適用（推奨）
#   1.0: Early cutoffを実質無効化（時間を使い切るまで探索）
#
# 推奨値:
#   序盤・中盤: 0.5～0.7（時間を有効活用しつつ、高スコアで節約）
#   終盤: 0.2～0.4（確実な手を見つけたら早期終了）
EARLY_CUTOFF_TIME_THRESHOLD=0.5


# ----------------------------------------------------
# メモリ設定
# ----------------------------------------------------

# TRANSPOSITION_TABLE_SIZE: 置換表のサイズ（エントリ数）
# 大きいほど探索が効率的になりますが、メモリを消費します。
# 推奨値: 65536～1048576
#   65536 (64K):   省メモリ（約5MB）
#   262144 (256K): 標準（約20MB）
#   1048576 (1M):  大容量（約80MB）
TRANSPOSITION_TABLE_SIZE=262144


# ----------------------------------------------------
# 学習設定
# ----------------------------------------------------

# LEARNING_MODE: 学習モード
# PRACTICE: 練習モード（対戦結果から学習し、スコアを更新）
# PRODUCTION: 本番モード（学習データを使用するが更新しない）
#
# 使い分け:
#   PRACTICE: トレーニング、AI同士の対戦、改善テスト
#   PRODUCTION: 大会、本番環境、安定した強さを維持
LEARNING_MODE=PRACTICE


# ----------------------------------------------------
# デバッグ設定
# ----------------------------------------------------

# DEBUG_MODE: 基本デバッグモード（レガシー設定）
# 0: 無効
# 1: 有効
# 推奨値: 0
DEBUG_MODE=0

# DEBUG_LEVEL: 詳細デバッグレベル
# 0: 出力なし（本番環境）
# 1: 基本情報（探索の進捗、最善手）
# 2: 詳細情報（盤面状態、統計、評価詳細）
# 3: 超詳細（全ノード、カットオフ、TT詳細）
#
# 推奨値:
#   0: 本番環境、大会
#   1: 通常の開発・テスト
#   2: 詳細なデバッグ
#   3: 問題解析（非常に多くの出力）
DEBUG_LEVEL=1

# DEBUG_LEVEL=0


# ====================================================
# 注意事項
# ====================================================
#
# 1. SEARCH_DEPTHを大きくする場合は、TIMEOUT_MSも増やすこと
#    （深い探索には時間がかかる）
#
# 2. メモリが限られている環境では、TRANSPOSITION_TABLE_SIZEを
#    小さくすること
#
# 3. 学習を無効にしたい場合は、LEARNING_MODE=PRODUCTION に設定
#
# 4. デバッグ出力が多すぎる場合は、DEBUG_LEVELを下げること
#
# 5. 設定を変更した後は、AIを再起動すること
